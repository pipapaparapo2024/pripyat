    $file = __DIR__ . '/core/models/'.strtolower($class_name).'.php';
    if(file_exists($file)) include($file);
});

$registry = new Registry();
$link = new mysqli($registry['server'], $registry['user'], $registry['pass'], $registry['db'], 3306);
if($link->connect_error){ die('Connect error: ' . $link->connect_error); }
$link->set_charset('utf8mb4');

$results = [];

$sql1 = "CREATE TABLE IF NOT EXISTS `reward_links` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT,
  `code` VARCHAR(64) NOT NULL,
  `reward` TEXT NOT NULL,
  `label` VARCHAR(255) DEFAULT NULL,
  `created_at` INT UNSIGNED NOT NULL DEFAULT 0,
  `expires_at` INT UNSIGNED DEFAULT NULL,
  `active` TINYINT UNSIGNED NOT NULL DEFAULT 1,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_code` (`code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4";

    $results[] = "OK    reward_links создана (или уже существовала)";

    $results[] = "ERROR " . $link->error;
}

$sql2 = "CREATE TABLE IF NOT EXISTS `reward_link_claims` (

  `link_id` INT UNSIGNED NOT NULL,
  `user_id` BIGINT UNSIGNED NOT NULL,
  `claimed_at` INT UNSIGNED NOT NULL DEFAULT 0,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_link_user` (`link_id`, `user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4";

if($link->query($sql2)){
    $results[] = "OK    reward_link_claims создана (или уже существовала)";
